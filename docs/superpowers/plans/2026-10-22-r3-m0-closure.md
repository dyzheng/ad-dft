- [ ] **Step 3: 修 `dion_full` 的块布局**

```python
def dion_full(psp: Pseudo, spec_idx) -> jnp.ndarray:
    """(nat*nproj, nat*nproj) 的块对角 D。

    关键：块索引必须放在 (i, p, j, q)，reshape 后全局行=(i,p)、列=(j,q)。
    不能写成 (i, j, p, q) —— 那会把不同原子的块交错成非 Hermitian 矩阵。
    """
    si = jnp.asarray(spec_idx)
    nat = int(si.shape[0])
    npr = psp.nproj
    d = jnp.zeros((nat, npr, nat, npr), dtype=psp.dion.dtype)
    for i in range(nat):
        d = d.at[i, :, i, :].set(psp.dion[si[i]])
    return d.reshape(nat * npr, nat * npr)
```

```python
def test_mp_2x2x2_is_in_first_bz():
```

```python
class KMesh(NamedTuple):
    kfrac: jnp.ndarray   # (nk,3) 以倒格矢为基的分数坐标
    weights: jnp.ndarray # (nk,) 和为 1

    def cartesian(self, lattice) -> jnp.ndarray:
        """k_cart = kfrac @ (2π inv(lattice).T)，单位 1/Bohr。"""
        bmat = 2.0 * jnp.pi * jnp.linalg.inv(lattice).T
        return self.kfrac @ bmat


def gamma() -> KMesh:
    return KMesh(kfrac=jnp.zeros((1, 3)), weights=jnp.ones((1,)))


def monkhorst_pack(n, shift=None) -> KMesh:
    n = tuple(int(x) for x in n)
    shift = (0.0, 0.0, 0.0) if shift is None else tuple(float(x) for x in shift)
    pts = []
    for i in range(n[0]):
        for j in range(n[1]):
            for k in range(n[2]):
                pts.append(((i + shift[0]) / n[0],
                            (j + shift[1]) / n[1],
                            (k + shift[2]) / n[2]))
    frac = np.asarray(pts, dtype=float)
    frac = np.where(frac >= 0.5, frac - 1.0, frac)
    return KMesh(
        kfrac=jnp.asarray(frac),
        weights=jnp.full((frac.shape[0],), 1.0 / frac.shape[0]),
    )
```

```python
def _nmax_on_axis(lattice, gcut2, kmax=0.0):
    lat = np.asarray(lattice, dtype=float)
    a_len = np.linalg.norm(lat, axis=1)
    m = np.ceil((np.sqrt(gcut2) + kmax) * a_len / (2.0 * np.pi)).astype(int)
    return int(m[0]), int(m[1]), int(m[2])


def build_miller(lattice, ecut, gamma_only=True, kpts=None):
    gcut2 = shell_radius2(ecut)
    lat = np.asarray(lattice, dtype=float)
    bmat = 2.0 * np.pi * np.linalg.inv(lat).T
    ks = np.zeros((1, 3)) if kpts is None else np.asarray(kpts)
    kmax = float(np.max(np.linalg.norm(ks, axis=1)))
    m1, m2, m3 = _nmax_on_axis(lat, gcut2, kmax)
    rng = np.stack(np.meshgrid(
        np.arange(-m1, m1 + 1), np.arange(-m2, m2 + 1),
        np.arange(-m3, m3 + 1), indexing="ij"), axis=-1).reshape(-1, 3)
    keep = np.zeros(len(rng), dtype=bool)
    for k in ks:
        kfrac = k @ np.linalg.inv(bmat)   # k_cart = kfrac @ bmat
        cart = (rng + kfrac) @ bmat
        keep |= np.sum(cart ** 2, axis=1) <= gcut2 * (1.0 + 1e-12)
    miller = rng[keep]
    g2 = np.sum((miller @ bmat) ** 2, axis=1)
    order = np.lexsort((miller[:, 2], miller[:, 1], miller[:, 0], g2))
    return jnp.asarray(miller[order], dtype=jnp.int32)


def make_basis(lattice, ecut, density_factor=4.0, kpts=None):
    return Basis(
        miller=build_miller(lattice, ecut, kpts=kpts),
        ecut=float(ecut),
        fft_shape=fft_shape(lattice, ecut, density_factor),
    )
```

```python
"""实球谐，l≤2，顺序与 ABACUS YlmReal 一致：每个 l 为 m=0,+1,-1,+2,-2..."""

import jax.numpy as jnp

_FOUR_PI = 4.0 * jnp.pi


def ylm_real(lmax: int, g):
    """g: (...,3) 笛卡尔；返回 (...,(lmax+1)^2)。"""
    g = jnp.asarray(g)
    x, y, z = g[..., 0], g[..., 1], g[..., 2]
    r2 = jnp.sum(g * g, axis=-1)
    rs = jnp.sqrt(jnp.where(r2 > 1e-30, r2, 1.0))
    out = [jnp.ones_like(rs) / jnp.sqrt(_FOUR_PI)]
    if lmax >= 1:
        c1 = jnp.sqrt(3.0 / _FOUR_PI) / rs
        out += [c1 * z, -c1 * x, -c1 * y]
    if lmax >= 2:
        c2 = jnp.sqrt(15.0 / _FOUR_PI) / (rs * rs)
        out += [c2 * (3.0 * z * z - r2) / 2.0,
                -c2 * x * z,
                -c2 * y * z,
                jnp.sqrt(15.0 / (16.0 * jnp.pi)) * (x * x - y * y) / (rs * rs),
                jnp.sqrt(15.0 / (4.0 * jnp.pi)) * x * y / (rs * rs)]
    return jnp.stack(out, axis=-1)
```

注意 l=2 的最后两个顺序要与 ABACUS `Am/Bm` 的 m=2 对应；若测试发现符号/顺序
不符，以 `math_ylmreal.cpp` 的 `rly[ic]` 输出为唯一标准，逐项调整，不允许改测试。

```python
    """返回 [(radial_index, l, m_index_local), ...]"""
```

```python
def projectors(psp, spec_idx, basis, cell, kpts):
    nat = int(spec_idx.shape[0])
    nh = int(psp.nh[int(spec_idx[0])])  # 本轮单 species
    r = positions(cell)
    omega = volume(cell.lattice)
    out = []
    for ik in range(int(kpts.shape[0])):
        gk = gvectors(basis, cell.lattice, kpts[ik])
        q = jnp.sqrt(jnp.sum(gk * gk, axis=-1))
        ylm = ylm_real(int(psp.lmax), gk)  # (nG,(lmax+1)^2)
        phase = jnp.exp(-1j * (gk @ r.T))
        rows = []
        for i in range(nat):
            si = int(spec_idx[i])
            bet = beta_q(psp, si, q)  # (nrad,nG)
            for ib, l, m in lm_table(psp.proj_l[si]):
                rows.append(bet[ib] * ylm[:, l * l + m] * phase[:, i])
        out.append(jnp.stack(rows, axis=0) / jnp.sqrt(omega))
    return jnp.stack(out, axis=0)
```

```python
def dion_full(psp, spec_idx):
    si = jnp.asarray(spec_idx)
    nat = int(si.shape[0])
    nh = int(psp.nh[int(si[0])])
    d = jnp.zeros((nat, nh, nat, nh), dtype=psp.dion.dtype)
    lm = lm_table(psp.proj_l[int(si[0])])
    for i in range(nat):
        for p, (ib, l, m) in enumerate(lm):
            for q, (jb, l2, m2) in enumerate(lm):
                if l == l2 and m == m2:
                    d = d.at[i, p, i, q].set(
                        psp.dion[int(si[i]), ib, jb])
    return d.reshape(nat * nh, nat * nh)
```

```python
def _read_v1(root):
    text = root.find("PP_HEADER").text
    zval = float(re.search(r"([-\d.]+)\s+Z valence", text).group(1))
    mesh = int(re.search(r"(\d+)\s+Number of points", text).group(1))
    mesh = mesh - (mesh % 2 == 0)
    nchi, nbeta = map(int, re.search(
        r"(\d+)\s+(\d+)\s+Number of Wavefunctions, Number of Projectors",
        text).groups())
    r = _nums(root.find("PP_MESH/PP_R"))[:mesh]
    rab = _nums(root.find("PP_MESH/PP_RAB"))[:mesh]
    vloc = _nums(root.find("PP_LOCAL"))[:mesh]
    lll, betar = [], []
    for el in root.findall("PP_NONLOCAL/PP_BETA"):
        vals = np.fromstring(el.text, sep=" ")
        idx, l, kb = int(vals[0]), int(vals[1]), int(vals[2])
        lll.append(l)
        betar.append(vals[3:3 + kb])
    # betar 的 kbeta 可能小于 mesh，后面补 0
    betar = np.stack([np.pad(b, (0, mesh - len(b))) for b in betar])
    dvals = np.fromstring(root.find("PP_NONLOCAL/PP_DIJ").text, sep=" ")
    nd = int(dvals[0])
    d = np.zeros((nbeta, nbeta))
    for k in range(nd):
        i, j, v = int(dvals[1 + 3 * k]), int(dvals[2 + 3 * k]), dvals[3 + 3 * k]
        d[i - 1, j - 1] = d[j - 1, i - 1] = v  # Ry
    nlcc_el = root.find("PP_NLCC")
    return dict(zval=zval, mesh=mesh, r=r, rab=rab, vloc_at=vloc,
                nbeta=nbeta, lll=np.asarray(lll), betar=betar, dion=d,
                nlcc=nlcc_el is not None,
                rho_atc=(_nums(nlcc_el)[:mesh] if nlcc_el is not None
                         else np.zeros(mesh)))
```

```python
def _simpson_weights(n):
    w = jnp.ones((n,), dtype=jnp.float64)
    w = w.at[1:n - 1:2].set(4.0)
    w = w.at[2:n - 1:2].set(2.0)
    return w / 3.0


def _sph_j(l, x):
    """j_l(x)，l≤2；x 可为 0。"""
    xs = jnp.where(jnp.abs(x) > 1e-10, x, 1.0)
    if l == 0:
        y = jnp.sin(xs) / xs
        return jnp.where(jnp.abs(x) > 1e-10, y, 1.0)
    if l == 1:
        y = jnp.sin(xs) / xs ** 2 - jnp.cos(xs) / xs
        return jnp.where(jnp.abs(x) > 1e-10, y, 0.0)
    y = (3.0 / xs ** 3 - 1.0 / xs) * jnp.sin(xs) - 3.0 * jnp.cos(xs) / xs ** 2
    return jnp.where(jnp.abs(x) > 1e-10, y, 0.0)


def beta_q(psp, spec_idx, q):
    if psp.kind == 0:  # 合成赝势：保留解析路径
        si = int(spec_idx)
        l = psp.proj_l[si].astype(q.dtype)[:, None]
        w = psp.proj_w[si][:, None]
        b = psp.proj_b[si][:, None]
        qq = jnp.maximum(q[None, :], 1e-12)
        return w * jnp.power(qq, l) * jnp.exp(-0.5 * (qq * b) ** 2)
    si = int(spec_idx)
    r, rab, betar = psp.r[si], psp.rab[si], psp.betar[si]
    w = _simpson_weights(r.shape[0])
    out = []
    for ib in range(psp.nrad):
        jl = _sph_j(int(psp.proj_l[si, ib]), q[:, None] * r[None, :])
        integral = jnp.sum(w[None, :] * rab[None, :] *
                           betar[ib][None, :] * r[None, :] * jl, axis=1)
        out.append(4.0 * jnp.pi * integral)  # 与 ABACUS tab 的前因子对齐
    return jnp.stack(out, axis=0)
```

```python
def vloc_q(psp, spec_idx, q):
    if psp.kind == 0:
        si = int(spec_idx)
        amp = psp.vloc_amp[si]
        sig = psp.vloc_sigma[si]
        return -amp * jnp.exp(-0.5 * (q * sig) ** 2)
    si = int(spec_idx)
    r, rab = psp.r[si], psp.rab[si]
    v = psp.vloc_at[si] / 2.0             # Ry -> Ha
    z = psp.zval[si]
    w = _simpson_weights(r.shape[0])
    qs = jnp.where(q > 1e-12, q, 1.0)
    aux = r * v + z * erf(r)
    integ = jnp.sum(w[None, :] * rab[None, :] * aux[None, :] *
                    jnp.sin(qs[:, None] * r[None, :]) / qs[:, None], axis=1)
    integ = integ - z * jnp.exp(-qs ** 2 / 4.0) / qs ** 2
    g0 = jnp.sum(w * rab * r * (r * v + z))
    integ = jnp.where(q > 1e-12, integ, g0)
    return 4.0 * jnp.pi * integ
```

```python
from pathlib import Path
import numpy as np


def parse_stru(path):
    text = Path(path).read_text(encoding="utf-8").splitlines()
    lat0, vecs = 1.0, []
    species_labels, pseudo_files = [], []
    species_per_atom, coords, magmoms = [], [], []
    mode = None
    i, n = 0, len(text)

    while i < n:
        line = text[i].split("#")[0].strip()
        if not line:
            i += 1
            continue
        if line.startswith("ATOMIC_SPECIES"):
            i += 1
            while i < n:
                s = text[i].split("#")[0].strip()
                if not s or s.startswith(("LATTICE_", "ATOMIC_POSITIONS")):
                    break
                parts = s.split()
                if len(parts) >= 3:
                    species_labels.append(parts[0])
                    pseudo_files.append(parts[2])
                i += 1
            continue
        if line.startswith("LATTICE_CONSTANT"):
            lat0 = float(line.split()[1])
            i += 1
            continue
        if line.startswith("LATTICE_VECTORS"):
            for j in range(3):
                vecs.append([float(x) for x in text[i + 1 + j].split()[:3]])
            i += 4
            continue
        if line.startswith("ATOMIC_POSITIONS"):
            parts = line.split()
            i += 1
            if len(parts) >= 2:
                mode = parts[1]
            else:
                while i < n:
                    s = text[i].split("#")[0].strip()
                    if not s:
                        i += 1
                        continue
                    mode = s.split()[0]
                    i += 1
                    break
            if mode not in ("Direct", "Cartesian"):
                raise NotImplementedError(f"未支持的坐标模式: {mode}")
            break
        i += 1

    while i < n:
        s = text[i].split("#")[0].strip()
        if not s:
            i += 1
            continue
        if s.startswith(("LATTICE_", "ATOMIC_")):
            break
        label = s.split()[0]
        if label not in species_labels:
            raise ValueError(f"ATOMIC_POSITIONS 中的元素未在 ATOMIC_SPECIES 定义: {label}")
        mag = float(text[i + 1].split()[0])
        count = int(float(text[i + 2].split()[0]))
        i += 3
        for _ in range(count):
            nums = [float(x) for x in text[i].split()[:3]]
            coords.append(nums)
            magmoms.append(mag)
            species_per_atom.append(label)
            i += 1

    if mode is None:
        raise ValueError("STRU 缺少 ATOMIC_POSITIONS 坐标模式")
    lat = np.asarray(vecs, dtype=float) * lat0          # Bohr
    if mode == "Direct":
        frac = np.asarray(coords, dtype=float)
    else:                                               # Cartesian，单位 lat0
        frac = (np.asarray(coords, dtype=float) * lat0) @ np.linalg.inv(lat)
    return dict(lattice=lat, frac=frac, species=species_per_atom,
                magmoms=np.asarray(magmoms), labels=species_labels,
                pseudo_files=pseudo_files)
```

```python
def run_case(case_dir, suffix="r3", timeout=600):

def _block_after(text, header):
    lines = text.splitlines()
    for i, line in enumerate(lines):
        if header in line:
            return lines[i + 2:i + 2 + 8]
    raise ValueError(f"找不到 {header}")


def parse_running_scf(path):
    text = Path(path).read_text(errors="ignore")
    # FINAL_ETOT_IS 的单位是 eV
    etot_ev = float(re.search(r"FINAL_ETOT_IS\s+([-\d.eE+]+)", text).group(1))
    # TOTAL-FORCE (eV/Angstrom)：行格式如 "Si1 0.0 0.0 -3.6e-05"
    f_lines = _block_after(text, "TOTAL-FORCE (eV/Angstrom)")
    f_rows = []
    for ln in f_lines:
        parts = ln.split()
        if len(parts) >= 4:
            try:
                f_rows.append([float(x) for x in parts[-3:]])
            except ValueError:
                pass
    forces = np.asarray(f_rows)

    # TOTAL-STRESS (kbar)：3 行、每行 3 个数
    s_lines = _block_after(text, "TOTAL-STRESS (kbar)")
    s_rows = []
    for ln in s_lines:
        parts = ln.split()
        if len(parts) >= 3:
            try:
                s_rows.append([float(x) for x in parts[:3]])
            except ValueError:
                pass
    stress = np.asarray(s_rows[:3])
    return dict(etot_ev=etot_ev, etot_ha=etot_ev / HA_TO_EV,
                forces=forces, stress=stress)
```

`TOTAL-FORCE` 的表头实际是 `TOTAL-FORCE (eV/Angstrom)`；`TOTAL-STRESS` 是
`TOTAL-STRESS (kbar)`。首次运行时先 `grep -n` 确认列数，再固定解析器。

```python
from jax.scipy.sparse.linalg import bicgstab


def implicit_grad(geo, static, state, loss, return_diagnostics=False,
                  mode="matrix_free", rtol=1e-10):
    n_star = state.rho_r
    g_n = jax.grad(lambda nn: loss(geo, static, state._replace(rho_r=nn)))(n_star)

    if mode == "dense" or static.basis.nfft <= 1500:
        J = build_jacobian(geo, static, state)
        A = jnp.eye(static.basis.nfft) - J
        lam = jnp.linalg.solve(A.T, g_n)
        aT = lambda y: A.T @ y
        info = 0
    else:
        apply = make_response(geo, static, state)
        apply_t = jax.linear_transpose(apply, n_star)
        def matvec(y):
            return y - apply_t(y)[0]
        lam, info = bicgstab(matvec, g_n, tol=rtol, atol=0.0, maxiter=200)
        if int(info) != 0:
            raise RuntimeError(f"adjoint BiCGSTAB 未收敛: info={int(info)}")
        aT = matvec

    d_theta_state = jax.grad(lambda gg: loss(gg, static, state))(geo)
    d_theta_map = jax.grad(
        lambda gg: jnp.sum(lam * scf_map(gg, static, n_star)))(geo)
    grad = _tree_add(d_theta_state, d_theta_map)
    if not return_diagnostics:
        return grad
    resid = jnp.linalg.norm(aT(lam) - g_n) / (jnp.linalg.norm(g_n) + 1e-30)
    return grad, {
        "adjoint_residual": float(resid),
        "solver_info": int(info),
        "lam_norm": float(jnp.linalg.norm(lam)),
        "g_norm": float(jnp.linalg.norm(g_n)),
    }
```

保留 `build_jacobian` 仅用于小体系测试/诊断，不再默认进入生产路径。

```python
def test_hartree_planewave_exact_is_the_m0_oracle():
    """I-2 的无截断 oracle：n=1+Σa cos(Gr) 的 V_H 有闭式解。"""
    from addft.geometry.basis import gvectors

    basis = make_basis(jnp.eye(3) * 6.0, 4.0)
    g = gvectors(basis, jnp.eye(3) * 6.0)
    shape = tuple(basis.fft_shape)
    idx = jnp.stack(jnp.meshgrid(*(jnp.arange(n) for n in shape),
                                 indexing="ij"), axis=-1).reshape(-1, 3)
    r = (idx / jnp.array(shape, dtype=jnp.float64)) @ (jnp.eye(3) * 6.0)
    rho = jnp.ones(basis.nfft)
    exact = jnp.zeros(basis.nfft)
    for amp, i in ((0.7, 1), (-0.4, 4), (0.25, 10)):
        g2i = float(jnp.sum(g[i] * g[i]))
        rho = rho + amp * jnp.cos(r @ g[i])
        exact = exact + 4.0 * jnp.pi * amp / g2i * jnp.cos(r @ g[i])
    v = hartree.hartree_potential_r(basis, jnp.eye(3) * 6.0, rho)
    assert float(jnp.max(jnp.abs(v - exact))) < 1e-10
```

## 4. 类型/接口一致性检查（写计划时已核对）

| 名称 | 本轮定义 | 使用处 |
|---|---|---|
| `KMesh.kfrac` | 分数坐标，不再叫 `kpts` | `system.make_static`、`test_kpoints` |
| `Pseudo.kind` | 0=合成，1=UPF；`from_upf` 置 1 | `vloc_q/beta_q` 分派 |
| `Pseudo.nrad` | 径向 β 个数 | `beta_q`、`dion_full` |
| `Pseudo.nh` | 每个 species 的 m 通道总数 `Σ(2l+1)` | `projectors`、`dion_full` |
| `dion_full` | `(nat,nh,nat,nh)` 的正确块布局 | `hamiltonian`、`nonlocal_energy` |
| `make_response` | 纯线性 `apply(dn)`，eigh 在外部 | `implicit_grad`、I-15 |
| `implicit_grad(mode="matrix_free")` | `dense` 仅诊断，`matrix_free` 生产 | `api.make_energy`、`delta_q_grad` |

**禁止事项：**

1. 不允许为了“让旧测试过”而改 `tests/` 的物理门槛；只允许改测试 bug（如 `[0]`）。
2. 不允许把 `eigh` 放进 `make_response.apply`。
3. 不允许在 `src/` 里出现 `*_to_ry`；Ry→Ha 只在 `upf.py` / `adapters/`。
4. 不允许静默接受 NLCC、多 species、未映射 INPUT 参数。
5. 不允许用 `jax.jacfwd(scf_map)` 穿过 `eigh` 替代响应算子。

## 5. 风险与降级路径

| 风险 | 触发信号 | 降级 |
|---|---|---|
| UPF v1 解析与 ABACUS 不一致 | `vloc_q`/`dion` 对照失败 | 先用 v2 ONCV（PBE），M0 报告改 PBE 并注明；v1 单独立任务 |
| libxc ABI/ctypes 不稳定 | `xc_func_init` 段错误/版本差异 | `ADDFT_XC=analytic` 跑解析 PZ；M0 报告标 `libxc=off`，不算作通过 K07 |
| m 通道改动使旧合成基线漂移 | 旧能量数字变化 | 这是修复；以不变量（Hermitian/ΣF/FD）为准，更新 README 数字 |
| BiCGSTAB 不收敛 | `info != 0` | 先 `restart`/增大 `maxiter`；仍失败回退 dense（仅小体系），并把 M0 性能项标未通过 |
| ABACUS `feat/deltap` 分支输出格式变化 | runner 正则抓不到 | 固定二进制提交 `c17cbac9b` 与仓库 HEAD `dd6294b58`；把解析器做成版本化 |
| 非 Γ 联合壳层与 ABACUS per-k 壳层差异 | 非 Γ 能量差随 k 增大 | 本轮 M0 只要求 Γ；非 Γ 只做自洽性，报告显式写“V1 才做 per-k 壳层” |
| 高对称 Si 的价带简并让 `jax.grad(scf_map)` 出现 NaN | 力为 NaN 或超大 | 能量对照用高对称 Si；力对照额外用 0.01 分数坐标的低对称畸变，并在报告中注明；彻底的 eigh custom_jvp 留到 K09/V1 |
| libxc 的 PZ 与 ABACUS 内置 PZ 差 >10 meV/atom | Si 能量差超标 | 先切 `ADDFT_XC=analytic` 做归因；若解析 PZ 对而 libxc 不对，锁 libxc ID 并报告，不混用两套 XC |
